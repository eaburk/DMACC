#!/bin/bash

# Prevent the OS installer from pausing or waiting for human interaction
export DEBIAN_FRONTEND=noninteractive

####-- Create a swap file so server can handle node & mongodb --####
fallocate -l 2G /swapfile
chmod 600 /swapfile
mkswap /swapfile
swapon /swapfile
echo '/swapfile none swap sw 0 0' | tee -a /etc/fstab

####-- Configure Prerequisites --####
apt-get update
apt-get install -y gnupg curl apt-transport-https ca-certificates software-properties-common

# Install nginx
apt install -y nginx
systemctl enable --now nginx

####-- Configure Nginx Directory Listing --####
cat << 'EOF' > /etc/nginx/sites-available/default
server {
  listen 80 default_server;
  listen [::]:80 default_server;
  root /var/www/html;
  index index.html index.htm;
  server_name _;
  location / {
    # Enable directory listing if index.html is missing
    autoindex on;
    # Optional: Optional tweaks for better readability
    autoindex_exact_size off;
    autoindex_localtime on;
    try_files $uri $uri/ =404;
  }
}
EOF

# Reload Nginx to apply the new configuration
systemctl reload nginx

# Install build essentials to support node packages
apt install -y build-essential
